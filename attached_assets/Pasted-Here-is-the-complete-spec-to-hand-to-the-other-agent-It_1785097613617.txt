Change 1 — spot-color-worker.ts: Fix marching squares boundary detection
The bug: marchingSquaresTrace uses === 1 to detect filled pixels. Fluorescent masks use value 255, so every boundary check fails — half of all edges are inserted with the wrong direction, producing thousands of disconnected fragments instead of clean closed shapes. White/Gloss masks use value 1 so they were unaffected.

Find these two if blocks inside marchingSquaresTrace and replace both comparisons:

// BEFORE (broken for any mask value other than 1):
if (above !== below) {
  if (above === 1) { ... }
  else { ... }
}
// ...
if (left !== right) {
  if (left === 1) { ... }
  else { ... }
}
// AFTER (works for value 0 or 255 or any non-zero):
if ((above > 0) !== (below > 0)) {
  if (above > 0) { ... }
  else { ... }
}
// ...
if ((left > 0) !== (right > 0)) {
  if (left > 0) { ... }
  else { ... }
}

Change 2 — spot-color-worker.ts: Replace traceMaskToInchPaths with the full smoothing pipeline
The bugs: (a) No pre-filter → thousands of noise islands processed unnecessarily. (b) No Douglas-Peucker → Chaikin receives staircase corners it can't smooth well. (c) Chaikin applied to hole contours → shrinks holes, collapsing thin ink rings (e.g. an outline around a character) into solid filled blobs. (d) 3 Chaikin iterations = 8× point multiplication, producing multi-MB content streams on complex designs.

Replace the existing traceMaskToInchPaths function and add the four helpers below it:

/** Shoelace signed area. Positive = CW winding in Y-down = outer contour.
 *  Negative = CCW = inner hole contour. Works in any units. */
function signedArea(pts: { x: number; y: number }[]): number {
  let sum = 0;
  for (let i = 0; i < pts.length; i++) {
    const j = (i + 1) % pts.length;
    sum += pts[i].x * pts[j].y - pts[j].x * pts[i].y;
  }
  return sum / 2;
}
/** Shoelace unsigned area (for filtering). */
function polygonArea(pts: { x: number; y: number }[]): number {
  return Math.abs(signedArea(pts));
}
/**
 * Douglas-Peucker simplification.
 * Collapses pixel-grid staircase runs into single diagonal segments
 * before Chaikin, so smoothing has clean corners to work with.
 */
function douglasPeucker(pts: { x: number; y: number }[], epsilon: number): { x: number; y: number }[] {
  if (pts.length < 3) return pts;
  const last = pts.length - 1;
  const ax = pts[0].x, ay = pts[0].y;
  const bx = pts[last].x, by = pts[last].y;
  const abLen = Math.sqrt((bx - ax) ** 2 + (by - ay) ** 2);
  let maxDist = 0, maxIdx = 0;
  for (let i = 1; i < last; i++) {
    const dist = abLen === 0
      ? Math.sqrt((pts[i].x - ax) ** 2 + (pts[i].y - ay) ** 2)
      : Math.abs((by - ay) * pts[i].x - (bx - ax) * pts[i].y + bx * ay - by * ax) / abLen;
    if (dist > maxDist) { maxDist = dist; maxIdx = i; }
  }
  if (maxDist > epsilon) {
    const left  = douglasPeucker(pts.slice(0, maxIdx + 1), epsilon);
    const right = douglasPeucker(pts.slice(maxIdx), epsilon);
    return [...left.slice(0, -1), ...right];
  }
  return [pts[0], pts[last]];
}
/**
 * Chaikin corner-cutting — 2 iterations converges toward a quadratic B-spline.
 * Only apply to OUTER contours (positive signed area). Never apply to holes —
 * Chaikin shrinks polygons, which collapses thin ink rings into solid blobs.
 */
function chaikinSmooth(pts: { x: number; y: number }[], iterations: number): { x: number; y: number }[] {
  let cur = pts;
  for (let iter = 0; iter < iterations; iter++) {
    const next: { x: number; y: number }[] = [];
    const n = cur.length;
    for (let i = 0; i < n; i++) {
      const a = cur[i], b = cur[(i + 1) % n];
      next.push({ x: 0.75 * a.x + 0.25 * b.x, y: 0.75 * a.y + 0.25 * b.y });
      next.push({ x: 0.25 * a.x + 0.75 * b.x, y: 0.25 * a.y + 0.75 * b.y });
    }
    cur = next;
  }
  return cur;
}
// Pre-filter in pixel space before any expensive processing.
// 200 sq-px ≈ a 14×14 region — anything smaller is noise.
const MIN_PIXEL_AREA = 200;
// Post-filter sanity check in inch space after smoothing.
const MIN_CONTOUR_AREA_SQ_IN = 2e-4;
function traceMaskToInchPaths(
  mask: Uint8Array,
  width: number,
  height: number,
  pixelsPerInch: number
): { x: number; y: number }[][] {
  const dpEpsilon = 1.0 / pixelsPerInch; // 1 pixel tolerance in inch space
  const rawPaths = marchingSquaresTrace(mask, width, height);
  const result: { x: number; y: number }[][] = [];
  for (const rawPath of rawPaths) {
    const collapsed = collapseCollinear(rawPath);
    if (collapsed.length < 3) continue;
    // Pre-filter: discard tiny noise contours before DP/Chaikin.
    if (Math.abs(signedArea(collapsed)) < MIN_PIXEL_AREA) continue;
    // Convert pixel coords → inches.
    const inchPts = collapsed.map(p => ({ x: p.x / pixelsPerInch, y: p.y / pixelsPerInch }));
    // Simplify staircase runs into diagonals.
    const simplified = douglasPeucker(inchPts, dpEpsilon);
    if (simplified.length < 3) continue;
    // Outer contours (CW, positive signed area) get Chaikin smoothing.
    // Inner/hole contours (CCW, negative) are left as-is — Chaikin would
    // shrink them, collapsing the hole and flooding the interior with ink.
    const sa = signedArea(simplified);
    const smoothed = sa > 0 ? chaikinSmooth(simplified, 2) : simplified;
    if (smoothed.length >= 3 && polygonArea(smoothed) >= MIN_CONTOUR_AREA_SQ_IN) {
      result.push(smoothed);
    }
  }
  return result;
}

Note: If polygonArea already exists in the file, remove the old version and use this one (with signedArea underneath it). collapseCollinear and marchingSquaresTrace are unchanged.

Change 3 — spot-color-vectors.ts: Fix the Y-down → Y-up coordinate transform
The bug: Both addSpotColorVectorsToPDF and addSpotColorVectorsFromMasksToPDF use the standard Y-up rotation matrix on path points that are in Y-down image space. This produces correct output at 0° rotation but growing mismatch at every other angle. The cross-term signs on sinR must be flipped.

Find the path transform inside both functions — it looks like this:

// BEFORE (wrong for Y-down input coords):
const relX = p.x - widthInches / 2;
const relY = p.y - heightInches / 2;
const rotX = relX * cosR - relY * sinR;
const rotY = relX * sinR + relY * cosR;
return {
  x: designCx + rotX,
  y: pageHeightInches - (designCy + rotY),
};

Replace with this in BOTH functions:

// AFTER (correct: relY is Y-down, so sinR cross-terms flip sign):
const relX = p.x - widthInches / 2;
const relY = p.y - heightInches / 2;
return {
  x: designCx + relX * cosR + relY * sinR,
  y: (pageHeightInches - designCy) + relX * sinR - relY * cosR,
};

The key difference: +relY * sinR in x (was minus), and −relY * cosR in y (was plus). The designCy is also moved outside the subtraction to keep the expression clear.
